(() => {
  const boot = window.AIRPORT_BOARD_BOOT || null;
  if (!boot) {
    return;
  }

  const tableEl = document.getElementById('fids-table');
  const previewContainer = document.getElementById('fids-preview');
  const rowsContainer = document.getElementById('fids-rows');
  const emptyStateEl = document.getElementById('fids-empty');
  const overlayEl = document.getElementById('fids-overlay');
  const refreshBarEl = document.getElementById('fids-refresh-bar');
  const refreshElapsedEl = document.getElementById('fids-refresh-elapsed');
  const refreshButtonEl = document.getElementById('fids-refresh-button');
  const filterButtonEl = document.getElementById('fids-filter-button');
  const editButtonEl = document.getElementById('fids-edit-button');
  const titleCodeTriggerEl = document.getElementById('fids-title-code-trigger');
  const loadButtonEl = document.getElementById('fids-load-button');
  const loadMessageEl = document.getElementById('fids-load-message');
  const turnstileContainer = document.getElementById('fids-turnstile');
  const appModalRootEl = document.getElementById('app-modal-root');
  const editorSbarRoot = document.getElementById('sbar-home-root');
  const editorSbarStagingEl = document.getElementById('fids-editor-sbar-staging');
  const localTimeEl = document.getElementById('localTime');
  const localDateEl = document.getElementById('localDate');
  const insightScheduledTodayEl = document.getElementById('fids-insight-scheduled-today');
  const insightScheduledTomorrowEl = document.getElementById('fids-insight-scheduled-tomorrow');
  const insightAirlinesTotalEl = document.getElementById('fids-insight-airlines-total');
  const insightAirlinesSubEl = document.getElementById('fids-insight-airlines-sub');
  const insightCityEl = document.getElementById('fids-insight-city');
  const insightCountryEl = document.getElementById('fids-insight-country');
  const insightTimezoneEl = document.getElementById('fids-insight-timezone');
  const insightTerminalsEl = document.getElementById('fids-insight-terminals');
  const insightAlternativeNamesEl = document.getElementById('fids-insight-alternative-names');

  const type = (boot.type || 'departures').toString().trim().toLowerCase() === 'arrivals' ? 'arrivals' : 'departures';
  const iata = (boot.iata || '').toString().trim().toUpperCase();
  const texts = boot.texts || {};
  const refreshCooldownMs = Number(boot.refreshCooldownMs || 30000);
  const timeFormatStorageKey = `airport_board_time_format_${type}_${iata.toLowerCase()}`;
  const boardSlugs = boot.boardSlugs || { en: { departures: 'departures', arrivals: 'arrivals' } };
  const defaultLocale = (boot.defaultLocale || 'en').toString().trim().toLowerCase();
  const locale = (boot.locale || defaultLocale || 'en').toString().trim().toLowerCase();
  const airportTimezone = (boot.airportTimezone || 'UTC').toString().trim() || 'UTC';
  const lastApiLoadCookieKey = `airport_board_last_load_${type}_${iata.toLowerCase()}`;
  const hasReferrer = typeof boot.hasReferrer === 'boolean'
    ? boot.hasReferrer
    : String(document.referrer || '').trim() !== '';
  const shouldAutoLoad = Boolean(boot.autoLoad) && hasReferrer;

  let use24h = (window.localStorage && window.localStorage.getItem(timeFormatStorageKey)) !== '12h';
  let fidsData = [];
  let filterOptions = {
    airlines: [],
    locations: [],
    terminals: [],
  };
  let filtersState = {
    airline: '',
    location: '',
    query: '',
    terminal: '',
  };
  let hasLoaded = false;
  let isLoading = false;
  let pendingFetch = false;
  let lastLiveFetchMs = 0;
  let lastErrorMessage = '';
  let editorModalOpen = false;
  let selectedEditorAirport = '';
  let selectedEditorType = type;
  let airportStatsSyncInFlight = false;
  let airportStatsLastSyncMs = 0;

  let turnstileWidgetId = null;
  let turnstileToken = '';
  let turnstileTokenIssuedAt = 0;
  let turnstileWidgetNeedsReset = false;
  let turnstileExecutionMode = 'render';
  let turnstileExecutionInFlight = false;
  let turnstileExecutionStartedAt = 0;
  const turnstileTokenMaxAgeMs = 110000;
  const turnstileRefreshLeadMs = 20000;
  const turnstileExecutionTimeoutMs = 20000;

  const getText = (key, fallback = '') => {
    const value = texts[key];
    return typeof value === 'string' && value.trim() !== '' ? value : fallback;
  };

  const normalize = (value) => String(value || '').trim().toLowerCase();
  const normalizeSearch = (value) => normalize(value).replace(/\s+/g, '');

  const escapeHtml = (value) => String(value || '')
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;');

  const toDisplay = (value, fallback = '--') => {
    const text = String(value || '').trim();
    return text === '' ? fallback : text;
  };

  const getCookieValue = (name) => {
    const value = ('; ' + document.cookie).split('; ' + name + '=');
    if (value.length < 2) return '';
    return decodeURIComponent(value.pop().split(';').shift() || '');
  };

  const setCookieValue = (name, value, maxAgeSeconds) => {
    document.cookie = `${name}=${encodeURIComponent(value)}; path=/; max-age=${maxAgeSeconds}`;
  };

  const getLastApiLoadTimestamp = () => {
    const raw = getCookieValue(lastApiLoadCookieKey);
    const parsed = parseInt(raw, 10);
    return Number.isFinite(parsed) ? parsed : 0;
  };

  const setLastApiLoadTimestamp = (timestamp) => {
    setCookieValue(lastApiLoadCookieKey, String(timestamp), 86400);
  };

  const getRefreshRemainingMs = () => {
    const lastTimestamp = getLastApiLoadTimestamp();
    if (!lastTimestamp) return 0;
    const elapsed = Date.now() - lastTimestamp;
    return elapsed >= refreshCooldownMs ? 0 : refreshCooldownMs - elapsed;
  };

  const formatElapsedSince = (timestampMs) => {
    const elapsedSeconds = Math.max(0, Math.floor((Date.now() - timestampMs) / 1000));
    if (elapsedSeconds < 60) {
      return `${elapsedSeconds}s ago`;
    }
    const minutes = Math.floor(elapsedSeconds / 60);
    const seconds = elapsedSeconds % 60;
    return `${minutes}m ${seconds}s ago`;
  };

  const parseTimeParts = (value) => {
    const text = String(value || '').trim();
    if (!text) return null;
    const match = text.match(/^(\d{1,2}):(\d{2})(?::(\d{2}))?\s*([AaPp][Mm])?$/);
    if (!match) return null;
    let hours = parseInt(match[1], 10);
    const minutes = parseInt(match[2], 10);
    if (!Number.isFinite(hours) || !Number.isFinite(minutes)) return null;
    const meridian = (match[4] || '').toUpperCase();
    if (meridian === 'PM' && hours < 12) hours += 12;
    if (meridian === 'AM' && hours === 12) hours = 0;
    return { hours, minutes };
  };

  const formatHoursMinutes = (hours, minutes) => {
    const safeHours = Number.isFinite(hours) ? Math.max(0, Math.min(23, Math.floor(hours))) : 0;
    const safeMinutes = Number.isFinite(minutes) ? Math.max(0, Math.min(59, Math.floor(minutes))) : 0;
    const paddedMinutes = String(safeMinutes).padStart(2, '0');
    if (use24h) {
      return `${String(safeHours).padStart(2, '0')}:${paddedMinutes}`;
    }
    const suffix = safeHours >= 12 ? 'PM' : 'AM';
    const twelveHour = safeHours % 12 || 12;
    return `${twelveHour}:${paddedMinutes} ${suffix}`;
  };

  const formatTimeText = (value) => {
    const text = String(value || '').trim();
    if (!text) return '';
    const parts = parseTimeParts(text);
    if (parts) {
      return formatHoursMinutes(parts.hours, parts.minutes);
    }
    const parsed = new Date(text);
    if (!Number.isNaN(parsed.getTime())) {
      return formatHoursMinutes(parsed.getHours(), parsed.getMinutes());
    }
    return text;
  };

  const formatDateTime = (dateValue, timeValue) => {
    const dateText = String(dateValue || '').trim();
    const timeText = formatTimeText(timeValue);
    if (dateText && timeText) return `${dateText} ${timeText}`;
    if (dateText) return dateText;
    return timeText;
  };

  const getDisplayTime = (flight) => flight.scheduledTime || flight.estimatedTime || flight.actualTime || '';

  const getFlightNumber = (flight) => {
    const primary = `${flight.airlineCode || ''}${flight.flightNumber || ''}`.trim();
    if (primary) return primary;
    const operating = `${flight.operatingAirlineCode || ''}${flight.operatedFlightNumber || ''}`.trim();
    if (operating) return operating;
    return String(flight.flightId || '').trim();
  };

  const normalizeIataCode = (value) => {
    const code = String(value || '').trim().toUpperCase();
    return /^[A-Z]{3}$/.test(code) ? code : '';
  };

  const normalizeFlightSlug = (value) => String(value || '').trim().toUpperCase().replace(/[^A-Z0-9]/g, '');

  const buildLiveTrackUrl = (flight) => {
    const flightNumber = normalizeFlightSlug(getFlightNumber(flight));
    const cityInfo = getCityAndCode(flight);
    const relatedAirportCode = normalizeIataCode(cityInfo.code);
    const boardAirportCode = normalizeIataCode(iata);

    if (!flightNumber || !relatedAirportCode || !boardAirportCode) {
      return '';
    }

    const from = type === 'arrivals' ? relatedAirportCode : boardAirportCode;
    const to = type === 'arrivals' ? boardAirportCode : relatedAirportCode;

    return `/${flightNumber}-${from}-${to}`;
  };

  const getStatusText = (flight) => (
    flight.remarksTranslated
    || flight.remarksWithTime
    || flight.remarks
    || flight.status
    || flight.statusCode
    || flight.remarksCode
    || ''
  );

  const getCityAndCode = (flight) => {
    if (type === 'arrivals') {
      return {
        city: flight.originFamiliarName || flight.originCity || flight.originAirportName || flight.origin || '',
        code: flight.originAirportCode || flight.origin || '',
        airport: flight.originAirportName || '',
      };
    }

    return {
      city: flight.destinationFamiliarName || flight.destinationCity || flight.destinationAirportName || flight.destination || '',
      code: flight.destinationAirportCode || flight.destination || '',
      airport: flight.destinationAirportName || '',
    };
  };

  const getStatusClass = (flight, message) => {
    const text = normalize(message);
    const remarksCode = String(flight.remarksCode || '').trim().toUpperCase();
    const delayedValue = typeof flight.delayed === 'string'
      ? flight.delayed.trim().toLowerCase()
      : flight.delayed;
    const isDelayedFlag = delayedValue === true || delayedValue === 1 || delayedValue === '1' || delayedValue === 'true';

    if (remarksCode === 'CANCELLED' || remarksCode === 'DIVERTED' || text.includes('cancel') || text.includes('divert')) {
      return 'status-danger';
    }
    if (remarksCode.includes('ON_TIME') || text.includes('on-time') || text.includes('on time') || text.includes('arrived')) {
      return 'status-success';
    }
    if (isDelayedFlag || remarksCode.includes('DELAY') || text.includes('delay') || text.includes('late')) {
      return 'status-warning';
    }
    if (text.includes('estimate') || text.includes('boarding') || text.includes('departed')) {
      return 'status-info';
    }
    return 'status-muted';
  };

  const buildBoardUrlForSelection = (code, boardType, selectedLocale) => {
    const safeCode = String(code || '').trim().toUpperCase();
    if (!/^[A-Z]{3}$/.test(safeCode)) return null;
    const safeLocale = String(selectedLocale || defaultLocale || 'en').trim().toLowerCase();
    const localeSlugs = boardSlugs[safeLocale] || boardSlugs[defaultLocale] || { departures: 'departures', arrivals: 'arrivals' };
    const safeType = boardType === 'arrivals' ? 'arrivals' : 'departures';
    return `/${localeSlugs[safeType]}/${safeCode}`;
  };

  const setLoadMessage = (message) => {
    if (loadMessageEl) {
      loadMessageEl.textContent = message || '';
    }
  };

  const syncOverlayVisibility = () => {
    if (!overlayEl) return;
    overlayEl.classList.toggle('is-hidden', hasLoaded && !pendingFetch);
  };

  const updateRefreshBar = () => {
    if (!refreshBarEl || !refreshElapsedEl || !refreshButtonEl) return;
    refreshBarEl.hidden = !hasLoaded;
    if (!hasLoaded) return;

    const latestFetch = lastLiveFetchMs || getLastApiLoadTimestamp() || Date.now();
    refreshElapsedEl.textContent = `Updated ${formatElapsedSince(latestFetch)}`;

    const remainingMs = getRefreshRemainingMs();
    const waitingForVerification = pendingFetch && !isTurnstileTokenValid();

    if (isLoading) {
      refreshButtonEl.disabled = true;
      refreshButtonEl.textContent = getText('refreshingBoard', 'Refreshing board...');
      return;
    }

    if (waitingForVerification) {
      refreshButtonEl.disabled = true;
      refreshButtonEl.textContent = getText('confirmingHuman', 'Confirming you are human...');
      return;
    }

    if (remainingMs > 0) {
      refreshButtonEl.disabled = true;
      refreshButtonEl.textContent = getText('refreshAvailableIn', 'Refresh board in :seconds s').replace(':seconds', String(Math.ceil(remainingMs / 1000)));
      return;
    }

    refreshButtonEl.disabled = false;
    refreshButtonEl.textContent = getText('refreshBoard', 'Refresh board');
  };

  const recoverStalledVerification = () => {
    if (!turnstileExecutionInFlight || !turnstileExecutionStartedAt) return false;
    const elapsedMs = Date.now() - turnstileExecutionStartedAt;
    if (elapsedMs < turnstileExecutionTimeoutMs) return false;

    turnstileExecutionInFlight = false;
    turnstileExecutionStartedAt = 0;
    if (window.turnstile && turnstileWidgetId !== null) {
      try {
        window.turnstile.reset(turnstileWidgetId);
      } catch (error) {
        // Ignore reset failures.
      }
    }
    if (pendingFetch) {
      pendingFetch = false;
      lastErrorMessage = 'Verification timed out. Please try again.';
    }
    return true;
  };

  const isTurnstileTokenValid = () => {
    if (!turnstileToken || !turnstileTokenIssuedAt) return false;
    return (Date.now() - turnstileTokenIssuedAt) < turnstileTokenMaxAgeMs;
  };

  const clearTurnstileToken = () => {
    turnstileToken = '';
    turnstileTokenIssuedAt = 0;
  };

  const invalidateTurnstileToken = () => {
    clearTurnstileToken();
    turnstileWidgetNeedsReset = true;
  };

  const resetTurnstileState = () => {
    clearTurnstileToken();
    turnstileWidgetNeedsReset = false;
    pendingFetch = false;
    turnstileExecutionInFlight = false;
    turnstileExecutionStartedAt = 0;
    if (window.turnstile && turnstileWidgetId !== null) {
      try {
        window.turnstile.reset(turnstileWidgetId);
      } catch (error) {
        // Ignore reset failures.
      }
    }
  };

  const requestTurnstileToken = (reason = 'background') => {
    if (turnstileExecutionMode !== 'execute') return;
    if (turnstileWidgetId === null || !window.turnstile || isTurnstileTokenValid()) return;

    if (turnstileExecutionInFlight) {
      if (reason === 'user') {
        turnstileExecutionInFlight = false;
        turnstileExecutionStartedAt = 0;
        try {
          window.turnstile.reset(turnstileWidgetId);
        } catch (error) {
          // Ignore reset failures.
        }
      }
      recoverStalledVerification();
      if (turnstileExecutionInFlight) return;
    }

    try {
      if (turnstileWidgetNeedsReset) {
        window.turnstile.reset(turnstileWidgetId);
        turnstileWidgetNeedsReset = false;
      }
      turnstileExecutionInFlight = true;
      turnstileExecutionStartedAt = Date.now();
      window.turnstile.execute(turnstileWidgetId);
    } catch (error) {
      turnstileExecutionInFlight = false;
      turnstileExecutionStartedAt = 0;
    }
  };

  const maintainTurnstileToken = () => {
    recoverStalledVerification();
    if (pendingFetch) return;
    if (turnstileExecutionMode !== 'execute') return;
    if (turnstileWidgetId === null || !window.turnstile || turnstileExecutionInFlight) return;
    if (!turnstileToken || !turnstileTokenIssuedAt) {
      requestTurnstileToken('background');
      return;
    }

    const tokenAgeMs = Date.now() - turnstileTokenIssuedAt;
    const refreshAtMs = turnstileTokenMaxAgeMs - turnstileRefreshLeadMs;
    if (tokenAgeMs >= refreshAtMs) {
      invalidateTurnstileToken();
      requestTurnstileToken('background');
    }
  };

  const initTurnstile = () => {
    if (!turnstileContainer) return false;
    if (turnstileWidgetId !== null) return true;
    const siteKey = turnstileContainer.dataset.sitekey || '';
    if (!siteKey) {
      turnstileContainer.classList.add('is-hidden');
      return false;
    }
    if (!window.turnstile) return false;

    turnstileExecutionMode = turnstileContainer.dataset.execution || 'render';
    turnstileWidgetId = window.turnstile.render(turnstileContainer, {
      sitekey: siteKey,
      theme: turnstileContainer.dataset.theme || 'light',
      size: turnstileContainer.dataset.size || 'flexible',
      appearance: turnstileContainer.dataset.appearance || 'always',
      execution: turnstileExecutionMode,
      callback: window.onTurnstileSuccess,
      'expired-callback': window.onTurnstileExpired,
      'error-callback': window.onTurnstileError,
    });
    return true;
  };

  const updateLoadButton = () => {
    if (!loadButtonEl) return;

    recoverStalledVerification();
    syncOverlayVisibility();
    updateRefreshBar();

    const remainingMs = getRefreshRemainingMs();
    const hasToken = isTurnstileTokenValid();
    const turnstileReady = !turnstileContainer || !!(turnstileContainer.dataset.sitekey || '');
    const waitingForVerification = pendingFetch && !hasToken;

    loadButtonEl.textContent = getText('loadButton', 'Load live board');
    loadButtonEl.disabled = isLoading || remainingMs > 0 || !turnstileReady || waitingForVerification;

    if (!turnstileReady) {
      setLoadMessage(getText('verificationUnavailable', 'Verification unavailable.'));
      return;
    }
    if (isLoading) {
      setLoadMessage(getText('loadingLiveData', 'Loading live data...'));
      return;
    }
    if (remainingMs > 0) {
      setLoadMessage(getText('refreshAvailableIn', 'Refresh board in :seconds s').replace(':seconds', String(Math.ceil(remainingMs / 1000))));
      return;
    }
    if (waitingForVerification) {
      setLoadMessage(getText('confirmingHuman', 'Confirming you are human...'));
      return;
    }
    if (lastErrorMessage) {
      setLoadMessage(lastErrorMessage);
      return;
    }
    setLoadMessage(getText('loadHint', 'Click load to verify and fetch live data.'));
  };

  const updateCooldownUi = () => {
    updateLoadButton();
    updateRefreshBar();
  };

  const buildAirlineOptionsFromFlights = (flights) => {
    const values = [];
    const seen = new Set();
    (Array.isArray(flights) ? flights : []).forEach((flight) => {
      const code = String(flight.airlineCode || '').trim().toUpperCase();
      const name = String(flight.airlineName || flight.airline || '').trim();
      const label = name && code ? `${name} (${code})` : (name || code);
      if (!label) return;
      const key = normalize(label);
      if (seen.has(key)) return;
      seen.add(key);
      values.push(label);
    });
    return values.sort((a, b) => a.localeCompare(b, undefined, { numeric: true, sensitivity: 'base' }));
  };

  const buildLocationOptionsFromFlights = (flights) => {
    const values = [];
    const seen = new Set();
    (Array.isArray(flights) ? flights : []).forEach((flight) => {
      const cityInfo = getCityAndCode(flight);
      const city = String(cityInfo.city || '').trim();
      const code = String(cityInfo.code || '').trim().toUpperCase();
      const label = city && code ? `${city} (${code})` : (city || code);
      if (!label) return;
      const key = normalize(label);
      if (seen.has(key)) return;
      seen.add(key);
      values.push(label);
    });
    return values.sort((a, b) => a.localeCompare(b, undefined, { numeric: true, sensitivity: 'base' }));
  };

  const normalizeTerminalValue = (value) => String(value || '').trim().toUpperCase();

  const buildTerminalOptionsFromFlights = (flights) => {
    const values = [];
    const seen = new Set();
    (Array.isArray(flights) ? flights : []).forEach((flight) => {
      const terminal = normalizeTerminalValue(flight.terminal || '');
      if (!terminal) return;
      if (seen.has(terminal)) return;
      seen.add(terminal);
      values.push(terminal);
    });
    return values.sort((a, b) => a.localeCompare(b, undefined, { numeric: true, sensitivity: 'base' }));
  };

  const updateFilterOptionsFromMeta = (meta) => {
    const safeMeta = meta && typeof meta === 'object' ? meta : {};
    const airlinesFromMeta = Array.isArray(safeMeta.airlines) ? safeMeta.airlines : [];
    const locationsFromMeta = Array.isArray(safeMeta.destinations || safeMeta.origins || safeMeta.airports)
      ? (safeMeta.destinations || safeMeta.origins || safeMeta.airports)
      : [];
    const terminalsFromMeta = Array.isArray(safeMeta.terminals) ? safeMeta.terminals : [];

    filterOptions.airlines = airlinesFromMeta.length ? airlinesFromMeta : buildAirlineOptionsFromFlights(fidsData);
    filterOptions.locations = locationsFromMeta.length ? locationsFromMeta : buildLocationOptionsFromFlights(fidsData);
    filterOptions.terminals = terminalsFromMeta.length ? terminalsFromMeta : buildTerminalOptionsFromFlights(fidsData);

    updateFilterButtonLabel();
  };

  const hasActiveFilters = () => !!(
    String(filtersState.query || '').trim() !== ''
    || String(filtersState.airline || '').trim() !== ''
    || String(filtersState.location || '').trim() !== ''
    || String(filtersState.terminal || '').trim() !== ''
  );

  const getActiveFilterCount = () => {
    let count = 0;
    if (String(filtersState.query || '').trim() !== '') count += 1;
    if (String(filtersState.airline || '').trim() !== '') count += 1;
    if (String(filtersState.location || '').trim() !== '') count += 1;
    if (String(filtersState.terminal || '').trim() !== '') count += 1;
    return count;
  };

  const updateFilterButtonLabel = () => {
    if (!filterButtonEl) return;
    const activeCount = getActiveFilterCount();
    filterButtonEl.classList.toggle('is-active', activeCount > 0);
    const labelEl = filterButtonEl.querySelector('.fids-filter-button-label');
    const badgeEl = filterButtonEl.querySelector('.fids-filter-button-badge');
    if (labelEl) {
      labelEl.textContent = getText('filterButton', 'Filters');
    }
    if (badgeEl) {
      badgeEl.hidden = activeCount <= 0;
      badgeEl.textContent = String(activeCount);
    }
  };

  const matchesFilters = (flight) => {
    const query = normalize(filtersState.query);
    const queryCompact = normalizeSearch(filtersState.query);
    const airlineFilter = normalize(filtersState.airline);
    const locationFilter = normalize(filtersState.location);
    const terminalFilter = normalizeTerminalValue(filtersState.terminal);

    const flightNumber = getFlightNumber(flight);
    const flightNumberCompact = normalizeSearch(flightNumber);
    const cityInfo = getCityAndCode(flight);
    const statusText = getStatusText(flight);

    if (query) {
      const haystack = [
        flightNumber,
        flight.airlineCode,
        flight.airlineName,
        cityInfo.city,
        cityInfo.code,
        cityInfo.airport,
        statusText,
        flight.gate,
        flight.terminal,
      ].map(normalize).join(' ');
      let matchesQuery = haystack.includes(query);
      if (!matchesQuery && queryCompact) {
        matchesQuery = flightNumberCompact.includes(queryCompact);
      }
      if (!matchesQuery) return false;
    }

    if (airlineFilter) {
      const airlineHaystack = [
        flight.airlineCode,
        flight.airlineName,
        flight.airline,
      ].map(normalize);
      const matchesAirline = airlineHaystack.some((value) => value === airlineFilter || value.includes(airlineFilter));
      if (!matchesAirline) return false;
    }

    if (locationFilter) {
      const locationHaystack = [
        cityInfo.city,
        cityInfo.code,
        cityInfo.airport,
      ].map(normalize);
      const matchesLocation = locationHaystack.some((value) => value === locationFilter || value.includes(locationFilter));
      if (!matchesLocation) return false;
    }

    if (terminalFilter && normalizeTerminalValue(flight.terminal || '') !== terminalFilter) {
      return false;
    }

    return true;
  };

  const getFilteredFlights = () => {
    if (!Array.isArray(fidsData) || !fidsData.length) return [];
    if (!hasActiveFilters()) return fidsData.slice();
    return fidsData.filter(matchesFilters);
  };

  const createBoardRowElement = (flight, modalList, modalIndex) => {
    const timeText = formatTimeText(getDisplayTime(flight)) || '--:--';
    const cityText = toDisplay(getCityAndCode(flight).city, '--');
    const flightText = toDisplay(getFlightNumber(flight), '--');
    const statusText = toDisplay(getStatusText(flight), '--');
    const gateText = toDisplay(flight.gate, '--');

    const row = document.createElement('div');
    row.className = 'fids-row is-clickable';
    row.setAttribute('role', 'button');
    row.setAttribute('tabindex', '0');
    row.setAttribute('aria-label', `Open details for ${flightText}`);
    row.addEventListener('click', () => openModalFromList(modalList, modalIndex));
    row.addEventListener('keydown', (event) => {
      if (event.key === 'Enter' || event.key === ' ') {
        event.preventDefault();
        openModalFromList(modalList, modalIndex);
      }
    });

    row.innerHTML = `
      <div class="cell-time">${escapeHtml(timeText)}</div>
      <div class="cell-city">${escapeHtml(cityText)}</div>
      <div class="cell-flight">${escapeHtml(flightText)}</div>
      <div class="cell-status ${escapeHtml(getStatusClass(flight, statusText))}">${escapeHtml(statusText)}</div>

        emptyStateEl.textContent = getText('emptyState', 'No flights found.');
        emptyStateEl.style.display = 'block';

      if (emptyStateEl) {
        emptyStateEl.textContent = getText('noFlightsFound', 'No flights found.');
        emptyStateEl.style.display = 'block';
      }
      return;
    }

    flights.forEach((flight, index) => {
      rowsContainer.appendChild(createBoardRowElement(flight, flights, index));
    });
  };

      <div class="detail-label">${escapeHtml(label)}</div>

  };

      : toDisplay(flight.codeshares || flight.codeShares || '', '');

    addDetailRowTo(container, getText('detailFlight', 'Flight'), getFlightNumber(flight), true);
    addDetailRowTo(container, getText('detailCity', type === 'arrivals' ? 'From' : 'To'), destinationText, true);
    addDetailRowTo(container, getText('detailAirport', 'Airport'), cityInfo.airport, false);
    addDetailRowTo(container, getText('detailScheduled', type === 'arrivals' ? 'Scheduled arrival' : 'Scheduled departure'), formatDateTime(flight.scheduledDate || flight.scheduledDateUtc, flight.scheduledTime), true);
    addDetailRowTo(container, getText('detailEstimated', type === 'arrivals' ? 'Estimated arrival' : 'Estimated departure'), formatDateTime(flight.estimatedDate || flight.estimatedDateUtc, flight.estimatedTime), false);
    addDetailRowTo(container, getText('detailActual', type === 'arrivals' ? 'Actual arrival' : 'Actual departure'), formatDateTime(flight.actualDate || flight.actualDateUtc, flight.actualTime), false);
    addDetailRowTo(container, getText('detailTerminal', 'Terminal'), flight.terminal, false);
    addDetailRowTo(container, getText('detailGate', 'Gate'), flight.gate, false);
    if (type === 'arrivals') {
      addDetailRowTo(container, getText('detailBaggage', 'Baggage'), flight.baggage, false);
    }
    addDetailRowTo(container, getText('detailCodeshares', 'Codeshares'), codeshareText, false);
    addDetailRowTo(container, getText('detailUpdated', 'Last updated'), flight.lastUpdated || flight.updatedAt || flight.lastUpdatedUtc, false);
  };

  const openModalFromList = (list, index) => {
    if (!window.AppModal || typeof window.AppModal.open !== 'function') return;
    const flights = Array.isArray(list) ? list : [];

    window.AppModal.open({
      title,
      bodyHtml: '<div id="fids-flight-detail" class="fids-modal-grid"></div>',
      headerActionsHtml: liveTrackUrl
        ? `<a class="fids-live-track-button" href="${escapeHtml(liveTrackUrl)}"><span class="fids-live-track-dot" aria-hidden="true"></span><span>Live track</span></a>`
        : '',
      footerHtml: `<div class="fids-editor-footer"><button type="button" class="app-modal-button" data-app-modal-close>${escapeHtml(getText('modalClose', 'Close'))}</button></div>`,
    });

    const container = document.getElementById('fids-flight-detail');
    renderFlightDetails(container, flight);
  };

  const updateFilterSelectState = (select) => {
    if (!select) return;
    select.classList.toggle('is-active', !!select.value);
  };

    emptyOption.textContent = placeholder;
    select.appendChild(emptyOption);

    (Array.isArray(items) ? items : []).forEach((item) => {

      option.value = String(item || '').trim();
      option.textContent = String(item || '').trim();

    });

  const getActiveFiltersSummaryText = () => {

    const total = Array.isArray(fidsData) ? fidsData.length : 0;
    const filtered = getFilteredFlights().length;
    if (!total) return '';

      .replace(':filtered', String(filtered))
      .replace(':total', String(total));
  };

  const openFiltersModal = () => {
    if (!window.AppModal || typeof window.AppModal.open !== 'function') return;

    window.AppModal.open({
      title: getText('filterTitle', 'Filters'),

        <form id="fids-filter-form" class="fids-filter-form" autocomplete="off">
          <div class="fids-filter-grid">
            <div>
              <label class="fids-editor-label" for="fids-filter-airline">${escapeHtml(getText('filterAirline', 'Airline'))}</label>
              <select id="fids-filter-airline" class="fids-filter-select"></select>
            </div>
            <div>
              <label class="fids-editor-label" for="fids-filter-location">${escapeHtml(getText('filterLocation', type === 'arrivals' ? 'Origin' : 'Destination'))}</label>
              <select id="fids-filter-location" class="fids-filter-select"></select>
            </div>
            <div>
              <label class="fids-editor-label" for="fids-filter-search">${escapeHtml(getText('filterSearch', 'Search'))}</label>
              <input id="fids-filter-search" class="fids-filter-input" type="text" placeholder="${escapeHtml(getText('filterSearchPlaceholder', 'Flight number, airline, city or gate'))}">
            </div>
            <div>
              <span class="fids-editor-label">${escapeHtml(getText('filterTerminal', 'Terminal'))}</span>
              <div id="fids-filter-terminal-buttons" class="fids-filter-terminal-buttons"></div>
            </div>
            <div>
              <span class="fids-editor-label">${escapeHtml(getText('timeFormat', 'Time format'))}</span>
              <div class="fids-filter-time-toggle">
                <span>${escapeHtml(getText('ampm', 'AM/PM'))}</span>
                <label class="fids-filter-switch">
                  <input type="checkbox" id="fids-filter-time-toggle"${use24h ? ' checked' : ''}>
                  <span class="fids-filter-slider"></span>
                </label>
                <span>${escapeHtml(getText('h24', '24H'))}</span>
              </div>
            </div>
            <div id="fids-filter-summary" class="fids-filter-summary"></div>
          </div>
        </form>
      `,
      footerHtml: `
        <div class="fids-filter-footer">
          <button type="button" class="app-modal-button" id="fids-filter-clear">${escapeHtml(getText('filterClear', 'Clear filters'))}</button>
          <button type="button" class="app-modal-button fids-editor-submit-btn" id="fids-filter-apply" data-app-modal-close>${escapeHtml(getText('filterApplyAll', 'Show all results'))}</button>
        </div>
      `,
    });

    const airlineSelectEl = document.getElementById('fids-filter-airline');
    const locationSelectEl = document.getElementById('fids-filter-location');
    const searchInputEl = document.getElementById('fids-filter-search');
    const terminalButtonsEl = document.getElementById('fids-filter-terminal-buttons');
    const timeToggleEl = document.getElementById('fids-filter-time-toggle');
    const clearButtonEl = document.getElementById('fids-filter-clear');
    const applyButtonEl = document.getElementById('fids-filter-apply');
    const summaryEl = document.getElementById('fids-filter-summary');

    const renderTerminalButtons = () => {
      if (!terminalButtonsEl) return;
      terminalButtonsEl.innerHTML = '';

      const addButton = (label, value, active) => {
        const button = document.createElement('button');
        button.type = 'button';
        button.className = `fids-filter-terminal-button${active ? ' is-active' : ''}`;
        button.textContent = label;
        button.addEventListener('click', () => {
          filtersState.terminal = value;
          renderTerminalButtons();
          renderRows();
          updateSummary();
        });
        terminalButtonsEl.appendChild(button);
      };

      addButton(getText('filterAllTerminals', 'All terminals'), '', !filtersState.terminal);
      filterOptions.terminals.forEach((terminal) => {
        addButton(terminal, terminal, normalizeTerminalValue(filtersState.terminal) === normalizeTerminalValue(terminal));
      });
    };

    const updateSummary = () => {
      if (summaryEl) {
        summaryEl.textContent = getActiveFiltersSummaryText();
      }
      if (applyButtonEl) {
        applyButtonEl.textContent = hasActiveFilters()
          ? getText('filterApplyCount', 'Show :count flights').replace(':count', String(getFilteredFlights().length))
          : getText('filterApplyAll', 'Show all results');
      }
    };

    fillFilterSelect(airlineSelectEl, filterOptions.airlines, getText('filterAnyAirline', 'Any airline'), filtersState.airline);
    fillFilterSelect(locationSelectEl, filterOptions.locations, getText('filterAnyLocation', type === 'arrivals' ? 'Any origin' : 'Any destination'), filtersState.location);
    if (searchInputEl) searchInputEl.value = String(filtersState.query || '');
    renderTerminalButtons();
    updateSummary();

    airlineSelectEl?.addEventListener('change', () => {
      filtersState.airline = airlineSelectEl.value || '';
      updateFilterSelectState(airlineSelectEl);
      renderRows();
      updateSummary();
    });

    locationSelectEl?.addEventListener('change', () => {
      filtersState.location = locationSelectEl.value || '';
      updateFilterSelectState(locationSelectEl);
      renderRows();
      updateSummary();
    });

    searchInputEl?.addEventListener('input', () => {
      filtersState.query = searchInputEl.value || '';
      renderRows();
      updateSummary();
    });

    timeToggleEl?.addEventListener('change', () => {
      use24h = !!timeToggleEl.checked;
      try {
        window.localStorage.setItem(timeFormatStorageKey, use24h ? '24h' : '12h');
      } catch (error) {
        // Ignore storage failures.
      }
      renderRows();
      tick();
      updateSummary();
    });

    clearButtonEl?.addEventListener('click', () => {
      filtersState = {
        airline: '',
        location: '',
        query: '',
        terminal: '',
      };
      if (searchInputEl) searchInputEl.value = '';
      if (airlineSelectEl) airlineSelectEl.value = '';
      if (locationSelectEl) locationSelectEl.value = '';
      updateFilterSelectState(airlineSelectEl);
      updateFilterSelectState(locationSelectEl);
      renderTerminalButtons();
      renderRows();
      updateSummary();
    });
  };

  const restoreEditorSearchbar = () => {
    if (editorSbarStagingEl && editorSbarRoot && !editorSbarStagingEl.contains(editorSbarRoot)) {
      editorSbarStagingEl.appendChild(editorSbarRoot);
      editorSbarStagingEl.hidden = true;
    }
    editorModalOpen = false;
    selectedEditorAirport = '';
    window.SBAR_HOME_ON_AIRPORT_SELECT = null;
  };

  const getSelectedAirportFromSearchbar = () => {
    const input = document.getElementById('sbar-home-frominput');
    if (!input) return '';
    const raw = String(input.value || '').toUpperCase();
    const wrappedMatch = raw.match(/\(([A-Z]{3})\)\s*$/);
    if (wrappedMatch) return wrappedMatch[1];
    const matches = raw.match(/\b[A-Z]{3}\b/g);
    return Array.isArray(matches) && matches.length ? matches[matches.length - 1] : '';
  };

  const openBoardEditor = () => {
    if (!window.AppModal || typeof window.AppModal.open !== 'function') return;

    selectedEditorAirport = '';
    selectedEditorType = type;

    window.AppModal.open({
      title: getText('editTitle', 'Edit board'),
      bodyHtml: `
        <form id="fids-editor-form" class="fids-editor-form">
          <div class="fids-editor-grid">
            <div>
              <span class="fids-editor-label">${escapeHtml(getText('editAirport', 'Airport'))}</span>
              <input type="hidden" id="fids-editor-airport" value="">
              <div id="fids-editor-sbar-slot" class="fids-editor-sbar-slot"></div>
            </div>
            <div>
              <span class="fids-editor-label">${escapeHtml(getText('editBoardType', 'Board type'))}</span>
              <div class="fids-editor-mode-options">
                <button type="button" class="fids-editor-mode-option is-active" data-editor-board-type="departures">${escapeHtml(getText('editDepartures', 'Departures'))}</button>
                <button type="button" class="fids-editor-mode-option" data-editor-board-type="arrivals">${escapeHtml(getText('editArrivals', 'Arrivals'))}</button>
              </div>
            </div>
          </div>
          <div id="fids-editor-error" class="fids-editor-error"></div>
        </form>
      `,
      footerHtml: `
        <div class="fids-editor-footer">
          <button type="button" class="app-modal-button" data-app-modal-close>${escapeHtml(getText('editCancel', 'Close'))}</button>
          <button type="button" class="app-modal-button fids-editor-submit-btn" id="fids-editor-submit">${escapeHtml(getText('editSave', 'Open board'))}</button>
        </div>
      `,
    });

    editorModalOpen = true;

    const slotEl = document.getElementById('fids-editor-sbar-slot');
    const airportInputEl = document.getElementById('fids-editor-airport');
    const submitEl = document.getElementById('fids-editor-submit');
    const errorEl = document.getElementById('fids-editor-error');
    const searchInputEl = document.getElementById('sbar-home-frominput');
    const modeButtons = Array.from(document.querySelectorAll('[data-editor-board-type]'));

    if (slotEl && editorSbarRoot) {
      slotEl.appendChild(editorSbarRoot);
      if (editorSbarStagingEl) {
        editorSbarStagingEl.hidden = true;
      }
    }

    if (typeof window.setSearchMode === 'function') {
      window.setSearchMode('oneways');
    }

    if (searchInputEl) {
      searchInputEl.value = '';
      searchInputEl.placeholder = getText('editAirportPlaceholder', 'Search airport or IATA');
    }

    const setEditorBoardType = (nextType) => {
      selectedEditorType = nextType === 'arrivals' ? 'arrivals' : 'departures';
      modeButtons.forEach((button) => {
        button.classList.toggle('is-active', button.dataset.editorBoardType === selectedEditorType);
      });
    };

    modeButtons.forEach((button) => {
      button.addEventListener('click', () => {
        setEditorBoardType(button.dataset.editorBoardType || 'departures');
      });
    });

    window.SBAR_HOME_ON_AIRPORT_SELECT = (selectedCode) => {
      selectedEditorAirport = String(selectedCode || '').trim().toUpperCase();
      if (airportInputEl) {
        airportInputEl.value = selectedEditorAirport;
      }
      if (errorEl) {
        errorEl.textContent = '';
      }
    };

    searchInputEl?.addEventListener('input', () => {
      if (airportInputEl) {
        airportInputEl.value = getSelectedAirportFromSearchbar();
      }
    });

    const submitEditor = () => {
      const selectedCode = String((airportInputEl?.value || selectedEditorAirport || getSelectedAirportFromSearchbar()) || '').trim().toUpperCase();
      if (!/^[A-Z]{3}$/.test(selectedCode)) {
        if (errorEl) {
          errorEl.textContent = getText('editInvalidAirport', 'Enter a valid IATA code.');
        }
        searchInputEl?.focus();
        return;
      }

      const url = buildBoardUrlForSelection(selectedCode, selectedEditorType, locale);
      if (!url) {
        if (errorEl) {
          errorEl.textContent = getText('editInvalidAirport', 'Enter a valid IATA code.');
        }
        return;
      }

      window.location.href = url;
    };

    submitEl?.addEventListener('click', submitEditor);
  };

  const tick = () => {
    const now = new Date();
    if (localTimeEl) {
      try {
        const formatted = new Intl.DateTimeFormat(use24h ? 'sv-SE' : 'en-US', {
          hour: use24h ? '2-digit' : 'numeric',
          minute: '2-digit',
          hour12: !use24h,
          timeZone: airportTimezone,
        }).format(now);
        localTimeEl.textContent = formatted.replace(/\u202f|\u00a0/g, ' ');
      } catch (error) {
        localTimeEl.textContent = now.toLocaleTimeString();
      }
    }

    if (localDateEl) {
      try {
        const formattedDate = new Intl.DateTimeFormat('en-US', {
          month: 'long',
          day: 'numeric',
          timeZone: airportTimezone,
        }).format(now);
        localDateEl.textContent = formattedDate;
      } catch (error) {
        localDateEl.textContent = '';
      }
    }
  };

  const renderAirportStats = (insights) => {
    if (!insights || typeof insights !== 'object') return;

    if (insightScheduledTodayEl) {
      insightScheduledTodayEl.textContent = String(insights.scheduled_today ?? '--');
    }
    if (insightScheduledTomorrowEl) {
      insightScheduledTomorrowEl.textContent = String(insights.scheduled_tomorrow ?? '--');
    }
    if (insightAirlinesTotalEl) {
      insightAirlinesTotalEl.textContent = String(insights.airlines_total ?? '--');
    }
    if (insightAirlinesSubEl) {
      const airlines = Array.isArray(insights.airlines_top) ? insights.airlines_top : [];
      insightAirlinesSubEl.textContent = airlines.length ? airlines.join(', ') : ' ';
    }
    if (insightCityEl) {
      insightCityEl.textContent = toDisplay(insights.city, '--');
    }
    if (insightCountryEl) {
      const country = String(insights.country || '').trim();
      const code = String(insights.country_code || '').trim();
      insightCountryEl.textContent = country ? `${country}${code ? ` (${code})` : ''}` : '--';
    }
    if (insightTimezoneEl) {
      insightTimezoneEl.textContent = toDisplay(insights.timezone, '--');
    }
    if (insightTerminalsEl) {
      insightTerminalsEl.textContent = Array.isArray(insights.terminals) && insights.terminals.length ? insights.terminals.join(', ') : '--';
    }
    if (insightAlternativeNamesEl) {
      const names = Array.isArray(insights.alternative_names) ? insights.alternative_names.filter(Boolean) : [];
      insightAlternativeNamesEl.innerHTML = names.length
        ? names.map((name) => `<div>${escapeHtml(name)}</div>`).join('')
        : '<div>--</div>';
    }
  };

  const syncAirportStatsFromApi = (forceSync = false) => {
    if (!boot.airportStatsApiUrl) return Promise.resolve(false);
    if (airportStatsSyncInFlight) return Promise.resolve(false);
    if (!forceSync && airportStatsLastSyncMs > 0 && (Date.now() - airportStatsLastSyncMs) < 120000) {
      return Promise.resolve(false);
    }

    airportStatsSyncInFlight = true;
    return fetch(boot.airportStatsApiUrl, { cache: 'no-store' })
      .then((response) => {
        if (!response.ok) {
          throw new Error('airport_stats_http_error');
        }
        return response.json();
      })
      .then((payload) => {
        if (payload && payload.insights) {
          renderAirportStats(payload.insights);
          airportStatsLastSyncMs = Date.now();
          return true;
        }
        return false;
      })
      .catch(() => false)
      .finally(() => {
        airportStatsSyncInFlight = false;
      });
  };

  const fetchData = () => {
    if (isLoading || !tableEl) return;

    const remainingMs = getRefreshRemainingMs();
    if (remainingMs > 0) {
      updateLoadButton();
      return;
    }

    if (turnstileContainer && turnstileWidgetId === null && !initTurnstile()) {
      lastErrorMessage = getText('verificationUnavailable', 'Verification unavailable.');
      pendingFetch = false;
      updateLoadButton();
      return;
    }

    if (turnstileContainer && !isTurnstileTokenValid()) {
      pendingFetch = true;
      lastErrorMessage = '';
      syncOverlayVisibility();
      requestTurnstileToken('user');
      updateLoadButton();
      return;
    }

    isLoading = true;
    lastErrorMessage = '';
    updateLoadButton();

    const apiBaseUrl = String(boot.apiBaseUrl || boot.apicBaseUrl || '').trim();
    if (!apiBaseUrl) {
      isLoading = false;
      lastErrorMessage = 'Unable to load live data.';
      updateLoadButton();
      updateRefreshBar();
      return;
    }

    const url = new URL(`${apiBaseUrl}/api/${encodeURIComponent(type)}`);
    url.searchParams.set('iata', iata);
    url.searchParams.set('origin', 'flightinformation.com');
    if (turnstileToken) {
      url.searchParams.set('cf-turnstile-response', turnstileToken);
    }

    invalidateTurnstileToken();

    fetch(url.toString(), { cache: 'no-store' })
      .then((response) => response.text().then((text) => {
        let data = null;
        if (text) {
          try {
            data = JSON.parse(text);
          } catch (error) {
            data = null;
          }
        }
        return {
          ok: response.ok,
          status: response.status,
          data,
        };
      }))
      .then((payload) => {
        isLoading = false;
        const data = payload ? payload.data : null;
        const meta = data && data.result ? data.result : {};
        const apiError = (data && data.error) ? data.error : (meta && meta.error ? meta.error : '');

        if (!payload.ok) {
          lastErrorMessage = apiError || (payload.status === 403 ? 'Verification expired. Please try again.' : 'Unable to load live data.');
          updateLoadButton();
          updateRefreshBar();
          maintainTurnstileToken();
          return;
        }

        if (apiError) {
          lastErrorMessage = apiError;
          updateLoadButton();
          updateRefreshBar();
          maintainTurnstileToken();
          return;
        }

        hasLoaded = true;
        syncOverlayVisibility();
        const fetchedAt = Date.now();
        setLastApiLoadTimestamp(fetchedAt);
        lastLiveFetchMs = fetchedAt;
        fidsData = Array.isArray(data?.fidsData) ? data.fidsData : (Array.isArray(meta?.fidsData) ? meta.fidsData : []);
        updateFilterOptionsFromMeta(meta);
        if (previewContainer) {
          previewContainer.style.display = 'none';
        }
        renderRows();
        updateLoadButton();
        updateRefreshBar();
        maintainTurnstileToken();
        syncAirportStatsFromApi(true);
      })
      .catch(() => {
        isLoading = false;
        lastErrorMessage = 'Unable to load live data.';
        updateLoadButton();
        updateRefreshBar();
        maintainTurnstileToken();
      });
  };

  if (loadButtonEl) {
    loadButtonEl.addEventListener('click', () => {
      if (loadButtonEl.disabled) return;
      fetchData();
    });
  }

  if (refreshButtonEl) {
    refreshButtonEl.addEventListener('click', () => {
      if (refreshButtonEl.disabled) return;
      fetchData();
    });
  }

  filterButtonEl?.addEventListener('click', openFiltersModal);
  editButtonEl?.addEventListener('click', openBoardEditor);
  titleCodeTriggerEl?.addEventListener('click', openBoardEditor);
  titleCodeTriggerEl?.addEventListener('keydown', (event) => {
    if (event.key === 'Enter' || event.key === ' ') {
      event.preventDefault();
      openBoardEditor();
    }
  });

  window.onTurnstileSuccess = (token) => {
    turnstileExecutionInFlight = false;
    turnstileExecutionStartedAt = 0;
    turnstileWidgetNeedsReset = true;
    turnstileToken = token || '';
    turnstileTokenIssuedAt = turnstileToken ? Date.now() : 0;
    if (pendingFetch) {
      pendingFetch = false;
      fetchData();
    }
    updateLoadButton();
    updateRefreshBar();
  };

  window.onTurnstileExpired = () => {
    const wasPending = pendingFetch;
    resetTurnstileState();
    if (wasPending) {
      lastErrorMessage = 'Verification expired. Please try again.';
    }
    updateLoadButton();
    updateRefreshBar();
  };

  window.onTurnstileError = () => {
    const wasPending = pendingFetch;
    resetTurnstileState();
    if (wasPending) {
      lastErrorMessage = 'Verification failed. Please try again.';
    }
    updateLoadButton();
    updateRefreshBar();
  };

  if (appModalRootEl) {
    appModalRootEl.addEventListener('click', (event) => {
      const target = event.target;
      if (!(target instanceof Element)) return;
      if (target.closest('[data-app-modal-close]') && editorModalOpen) {
        window.setTimeout(restoreEditorSearchbar, 0);
      }
    });
    document.addEventListener('keydown', (event) => {
      if (event.key === 'Escape' && editorModalOpen) {
        window.setTimeout(restoreEditorSearchbar, 0);
      }
    });
  }

  renderAirportStats({
    scheduled_today: insightScheduledTodayEl?.textContent || '--',
    scheduled_tomorrow: insightScheduledTomorrowEl?.textContent || '--',
    airlines_total: insightAirlinesTotalEl?.textContent || '--',
    airlines_top: insightAirlinesSubEl?.textContent ? insightAirlinesSubEl.textContent.split(',').map((item) => item.trim()).filter(Boolean) : [],
    city: insightCityEl?.textContent || '',
    country: insightCountryEl?.textContent || '',
    timezone: insightTimezoneEl?.textContent || '',
    terminals: insightTerminalsEl?.textContent ? insightTerminalsEl.textContent.split(',').map((item) => item.trim()).filter(Boolean) : [],
  });

  const initTimer = window.setInterval(() => {
    if (initTurnstile()) {
      window.clearInterval(initTimer);
      maintainTurnstileToken();
      updateLoadButton();
      if (shouldAutoLoad) {
        fetchData();
      }
    }
  }, 300);

  tick();
  window.setInterval(tick, 1000);
  window.setInterval(updateCooldownUi, 1000);
  window.setInterval(maintainTurnstileToken, 15000);
  syncAirportStatsFromApi(true);
  updateFilterButtonLabel();
  updateLoadButton();
})();
